GRAPH IMPLEMENTATION (LangGraph)
from langgraph.graph import StateGraph, END
from agents.router_agent import router_node
from agents.financial_agent import financial_node
from agents.comparison_agent import comparison_node
from agents.portfolio_agent import portfolio_node
from agents.synthesis_agent import synthesis_node

from state import ResearchState


def build_graph():

    graph = StateGraph(ResearchState)

    graph.add_node("router", router_node)
    graph.add_node("financial", financial_node)
    graph.add_node("comparison", comparison_node)
    graph.add_node("portfolio", portfolio_node)
    graph.add_node("synthesis", synthesis_node)

    graph.set_entry_point("router")

    graph.add_conditional_edges(
        "router",
        lambda state: state["route"],
        {
            "financial": "financial",
            "comparison": "comparison",
            "portfolio": "portfolio",
        }
    )

    graph.add_edge("financial", "synthesis")
    graph.add_edge("comparison", "synthesis")
    graph.add_edge("portfolio", "synthesis")

    graph.add_edge("synthesis", END)

    return graph.compile()
ROUTER NODE
def router_node(state: ResearchState):

    query = state["user_query"].lower()

    if "compare" in query:
        return {**state, "route": "comparison"}

    if "portfolio" in query:
        return {**state, "route": "portfolio"}

    return {**state, "route": "financial"}
FINANCIAL NODE
def financial_node(state: ResearchState):
